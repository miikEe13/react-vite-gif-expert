import { useState, useEffect, useRef } from 'react';
import { AddCategory, GifGrid } from './components';

export const GifExpertApp = () => {
    const [category, setCategory] = useState(''); // Changed to a single category (string)
    const containerRef = useRef(null); // Reference to the container for height calculation

    /**
     * Function to add a new category
     * @param {string} newCategory - New category to search
     */
    const onAddCategory = (newCategory) => {
        console.log('Adding category:', newCategory);
        if (category === newCategory) return; // Prevents duplicate searches
        setCategory(newCategory); // Updates the active category
    };

    /**
     * Function to remove a category
     * @param {*} categoryToRemove
     */
    const onRemoveCategory = () => {
        setCategory('');
    };

    /**
     * Send dynamic height to the parent using postMessage
     */
    const sendHeightToParent = () => {
        if (window.parent) {
            const height = containerRef.current.scrollHeight; // Calculate the total height
            console.log('Sending height to parent:', height);
            window.parent.postMessage({ type: 'iframeHeight', height }, '*'); // Change '*' to the parent domain
        }
    };

    /**
     * Use ResizeObserver to detect changes in the size of the container
     */
    useEffect(() => {
        const observer = new ResizeObserver((entries) => {
            console.log(entries);
            for (let entry of entries) {
                if (entry.target === containerRef.current) {
                    const newHeight = entry.contentRect.height; // New height of the container
                    sendHeightToParent(newHeight); // Send the height to the parent
                }
            }
        });

        if (containerRef.current) {
            observer.observe(containerRef.current); // Observe the container
        }

        return () => {
            if (containerRef.current) {
                observer.unobserve(containerRef.current); // Stop observing on unmount
            }
        };
    }, []); // Runs only on mount/unmount

    return (
        <div ref={containerRef} className='iframe-wrapper'>
            <AddCategory onNewCategory={onAddCategory} />
            <div className='container-grid'>
                <GifGrid
                    key={category} // Use the category as a unique key
                    category={category} // Pass the active category
                    onRemoveCategory={() => onRemoveCategory()} // Pass the remove function as a prop
                />
            </div>
        </div>
    );
};
